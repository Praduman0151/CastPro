cmake_minimum_required(VERSION 3.22.1)
project(castpro_native_engine LANGUAGES CXX)
add_library(castpro_native_engine SHARED NativeScreenEncoder.cpp)
find_library(log-lib log)
find_library(android-lib android)
find_library(mediandk-lib mediandk)
target_compile_features(castpro_native_engine PRIVATE cxx_std_17)
target_compile_options(castpro_native_engine PRIVATE -O2 -Wall -Wextra)
target_link_libraries(castpro_native_engine PRIVATE ${mediandk-lib} ${android-lib} ${log-lib})
